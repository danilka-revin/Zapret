#   # ultralytics тянет opencv-python с libGL; на безголовом роботе:
#   .venv/bin/pip uninstall -y opencv-python
#   .venv/bin/pip install "opencv-python-headless<5"
#
# Обучение:  .venv/bin/python scripts/train_vision.py --epochs 150 --imgsz 416
# Проверка (pytest из requirements-dev.txt):
#   .venv/bin/python -m pytest tests/test_vision.py
